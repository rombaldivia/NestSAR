def head_outputs(
    fusion_logits,
    joint_active,
    nonjoint_active,
    joint_target,
    nonjoint_target,
):

    (
        joint_logits,
        joint_prediction,

    ) = joint_active


    (
        nonjoint_logits,
        nonjoint_prediction,

    ) = nonjoint_active


    stream_logits = jnp.swapaxes(

        jnp.concatenate(

            (

                joint_logits[
                    None,
                    ...
                ],

                nonjoint_logits,
            ),

            axis=0,
        ),

        0,
        1,
    )


    fusion_weights = jax.nn.softmax(
        fusion_logits
    )


    logits = jnp.einsum(

        "s,bsc->bc",

        fusion_weights,

        stream_logits,
    )


    stream_prediction = jnp.swapaxes(

        jnp.concatenate(

            (

                joint_prediction[
                    None,
                    ...
                ],

                nonjoint_prediction,
            ),

            axis=0,
        ),

        0,
        1,
    )


    stream_target = jax.lax.stop_gradient(

        jnp.swapaxes(

            jnp.concatenate(

                (

                    joint_target[
                        None,
                        ...
                    ],

                    nonjoint_target,
                ),

                axis=0,
            ),

            0,
            1,
        )
    )


    return (

        logits,

        stream_logits,

        fusion_weights,

        stream_prediction,

        stream_target,
    )


def masked_head_loss(
    fusion_logits,
    joint_active,
    nonjoint_active,
    joint_target,
    nonjoint_target,
    labels,
    valid,
):

    (
        logits,
        stream_logits,
        fusion_weights,
        stream_prediction,
        stream_target,

    ) = head_outputs(

        fusion_logits,

        joint_active,
        nonjoint_active,

        joint_target,
        nonjoint_target,
    )


    valid = valid.astype(
        jnp.float32
    )


    denom = jnp.maximum(

        jnp.sum(
            valid
        ),

        1.0,
    )


    # --------------------------------------------------------------------------------------
    # FUSED CE
    # --------------------------------------------------------------------------------------

    one_hot = jax.nn.one_hot(

        labels,

        ns.CFG.num_classes,
    )


    smooth = (

        one_hot
        *
        (
            1.0
            -
            ns.CFG.label_smoothing
        )

        +

        ns.CFG.label_smoothing
        /
        ns.CFG.num_classes
    )


    fused_ce_each = optax.softmax_cross_entropy(

        logits,

        smooth,
    )


    fused_ce = (

        jnp.sum(

            fused_ce_each
            *
            valid
        )

        /

        denom
    )


    # --------------------------------------------------------------------------------------
    # AUX STREAM CE
    # [B,4,C]
    # --------------------------------------------------------------------------------------

    smooth_stream = smooth[
        :,
        None,
        :
    ]


    stream_ce_each = optax.softmax_cross_entropy(

        stream_logits,

        smooth_stream,
    )


    stream_ce_each = jnp.mean(

        stream_ce_each,

        axis=1,
    )


    stream_ce = (

        jnp.sum(

            stream_ce_each
            *
            valid
        )

        /

        denom
    )


    # --------------------------------------------------------------------------------------
    # PREDICTIVE
    #
    # prediction t predicts target t+1
    # --------------------------------------------------------------------------------------

    pred_now = stream_prediction[
        :,
        :,
        :-1,
        :
    ]


    target_next = jax.lax.stop_gradient(

        stream_target[
            :,
            :,
            1:,
            :
        ]
    )


    pred_each = jnp.mean(

        jnp.square(

            pred_now
            -
            target_next
        ),

        axis=(
            1,
            2,
            3,
        ),
    )


    predictive = (

        jnp.sum(

            pred_each
            *
            valid
        )

        /

        denom
    )


    total = (

        fused_ce

        +

        m4.AUX_STREAM_CE_WEIGHT
        *
        stream_ce

        +

        ns.CFG.predictive_loss_weight
        *
        predictive
    )


    predicted_class = jnp.argmax(

        logits,

        axis=-1,
    )


    correct = jnp.sum(

        (
            predicted_class
            ==
            labels
        ).astype(
            jnp.float32
        )

        *

        valid
    )


    # Stream correct counts.
    stream_pred_class = jnp.argmax(

        stream_logits,

        axis=-1,
    )


    stream_correct = jnp.sum(

        (
            stream_pred_class
            ==
            labels[
                :,
                None
            ]
        ).astype(
            jnp.float32
        )

        *

        valid[
            :,
            None
        ],

        axis=0,
    )


    aux = (

        fused_ce,
        stream_ce,
        predictive,

        correct,
        denom,

        fusion_weights,

        stream_correct,
    )


    return (
        total,
        aux,
    )


def head_backward(
    fusion_logits,
    joint_active,
    nonjoint_active,
    joint_target,
    nonjoint_target,
    labels,
    valid,
):

    return jax.value_and_grad(

        masked_head_loss,

        argnums=(
            0,
            1,
            2,
        ),

        has_aux=True,

    )(
        fusion_logits,

        joint_active,
        nonjoint_active,

        joint_target,
        nonjoint_target,

        labels,
        valid,
    )

)

def compute_gradient(p, batch_x, batch_y, valid, rng):
    mask_rng, dropout_root = jax.random.split(rng)
    cross_rng, joint_rng, nonjoint_rng = jax.random.split(dropout_root, 3)

    masked_x = regmask_step(batch_x, mask_rng)
    joint_xyz, nonjoint_xyz, geometry, stream_tokens = preprocess_step(masked_x)

    p_cross = p["cross_stream_mixer"]
    p_joint = p["stream_joint"]
    p_nonjoint = p["stream_nonjoint_bank"]
    p_fusion = p["stream_fusion_logits"]

    all_hints = cross_train_exe(p_cross, stream_tokens, cross_rng)
    joint_hints, nonjoint_hints = _split_hints(all_hints)

    joint_active, joint_target = joint_train_exe(
        p_joint, joint_xyz, geometry, joint_hints, joint_rng
    )
    nonjoint_active, nonjoint_target = nonjoint_train_exe(
        p_nonjoint, nonjoint_xyz, joint_xyz, geometry, nonjoint_hints, nonjoint_rng
    )

    ((loss, aux), (grad_fusion, cot_joint, cot_nonjoint)) = head_backward_exe(
        p_fusion,
        joint_active,
        nonjoint_active,
        joint_target,
        nonjoint_target,
        batch_y,
        valid,
    )

    grad_joint, cot_joint_hints = joint_vjp_exe(
        p_joint, joint_xyz, geometry, joint_hints, joint_rng, cot_joint
    )
    grad_nonjoint, cot_nonjoint_hints = nonjoint_vjp_exe(
        p_nonjoint,
        nonjoint_xyz,
        joint_xyz,
        geometry,
        nonjoint_hints,
        nonjoint_rng,
        cot_nonjoint,
    )

    # Reassemble the hint cotangents into [B,T,4,D] at each scale.
    cot_all_hints = tuple(
        jnp.concatenate(
            (
                cj[:, :, None, :],
                jnp.transpose(cn, (1, 2, 0, 3)),
            ),
            axis=2,
        )
        for cj, cn in zip(cot_joint_hints, cot_nonjoint_hints)
    )

    grad_cross = cross_vjp_exe(
        p_cross, stream_tokens, cross_rng, cot_all_hints
    )

    grads = {
        "cross_stream_mixer": grad_cross,
        "stream_joint": grad_joint,
        "stream_nonjoint_bank": grad_nonjoint,
        "stream_fusion_logits": grad_fusion,
    }

    if tree_numel(grads) != EXPECTED_PARAMS:
        raise RuntimeError(
            f"Gradient parameter count mismatch: {tree_numel(grads):,} != {EXPECTED_PARAMS:,}"
        )
    return loss, aux, grads

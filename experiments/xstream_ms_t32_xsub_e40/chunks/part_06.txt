 frame_blocks: int = 2
    chunk_blocks: int = 2
    clip_blocks: int = 2
    controller_blocks: int = 2
    chunk_size: int = 4
    clip_size: int = 8
    controller_rank: int = 32

    @nn.compact
    def __call__(self, x: jnp.ndarray, training: bool):
        if x.shape[1] % self.chunk_size:
            raise ValueError("frames must be divisible by chunk_size")
        if x.shape[1] % self.clip_size:
            raise ValueError("frames must be divisible by clip_size")

        streams = m4.build_four_streams(x)
        joint_xyz = streams["joint"]
        geometry = m4.geometric_features(joint_xyz)
        stream_tokens = _stack_stream_tokens(streams)

        cross_hints, cross_attention, cross_gates = CrossStreamMultiScaleHint(
            model_dim=self.model_dim,
            attention_dim=CROSS_DIM,
            heads=CROSS_HEADS,
            dropout=CROSS_DROPOUT,
            name="cross_stream_mixer",
        )(stream_tokens, training)
        joint_hints, nonjoint_hints = _split_hints(cross_hints)

        joint_out = StreamBackbone(
            stream_name="joint",
            num_classes=self.num_classes,
            model_dim=self.model_dim,
            memory_dim=self.memory_dim,
            dropout=self.dropout,
            memory_residual_scale=self.memory_residual_scale,
            initial_eta=self.initial_eta,
            initial_alpha=self.initial_alpha,
            frame_blocks=self.frame_blocks,
            chunk_blocks=self.chunk_blocks,
            clip_blocks=self.clip_blocks,
            controller_blocks=self.controller_blocks,
            chunk_size=self.chunk_size,
            clip_size=self.clip_size,
            controller_rank=self.controller_rank,
            use_part_attention=False,
            name="stream_joint",
        )(
            joint_xyz,
            joint_xyz,
            geometry,
            joint_hints[0],
            joint_hints[1],
            joint_hints[2],
            training,
        )

        nonjoint_xyz = jnp.stack(
            (
                streams["bone"],
                streams["joint_motion"],
                streams["bone_motion"],
            ),
            axis=0,
        )
        nonjoint_out = VectorizedNonJointBackbone(
            stream_name="bone",
            num_classes=self.num_classes,
            model_dim=self.model_dim,
            memory_dim=self.memory_dim,
            dropout=self.dropout,
            memory_residual_scale=self.memory_residual_scale,
            initial_eta=self.initial_eta,
            initial_alpha=self.initial_alpha,
            frame_blocks=self.frame_blocks,
            chunk_blocks=self.chunk_blocks,
            clip_blocks=self.clip_blocks,
            controller_blocks=self.controller_blocks,
            chunk_size=self.chunk_size,
            clip_size=self.clip_size,
            controller_rank=self.controller_rank,
            use_part_attention=False,
            name="stream_nonjoint_bank",
        )(
            nonjoint_xyz,
            joint_xyz,
            geometry,
            nonjoint_hints[0],
            nonjoint_hints[1],
            nonjoint_hints[2],
            training,
        )

        stream_logits = jnp.swapaxes(
            jnp.concatenate((joint_out["logits"][None, ...], nonjoint_out["logits"]), axis=0),
            0,
            1,
        )
        fusion_logits = self.param("stream_fusion_logits", nn.initializers.zeros, (NUM_STREAMS,))
        fusion_weights = jax.nn.softmax(fusion_logits)
        logits = jnp.einsum("s,bsc->bc", fusion_weights, stream_logits)

        stream_prediction = jnp.swapaxes(
            jnp.concatenate((joint_out["prediction"][None, ...], nonjoint_out["prediction"]), axis=0),
            0,
            1,
        )
        stream_motion_target = jnp.swapaxes(
            jnp.concatenate((joint_out["motion_target"][None, ...], nonjoint_out["motion_target"]), axis=0),
            0,
            1,
        )
        prediction = jnp.einsum("s,bstd->btd", fusion_weights, stream_prediction)
        motion_target = jnp.einsum("s,bstd->btd", fusion_weights, stream_motion_target)

        def average_metric(name: str):
            joint_mean = jnp.mean(joint_out[name])
            value = nonjoint_out[name]
            nonjoint_means = value if value.ndim == 1 else jnp.mean(
                value, axis=tuple(range(1, value.ndim))
            )
            values = jnp.concatenate((joint_mean[None], nonjoint_means.reshape((NUM_NONJOINT,))), axis=0)
            return jnp.mean(values)

        result = {
            "logits": logits,
            "stream_logits": stream_logits,
            "fusion_weights": fusion_weights,
            "prediction": prediction,
            "motion_target": motion_target,
            "stream_prediction": stream_prediction,
            "stream_motion_target": stream_motion_target,
            "memory_delta": jnp.mean(
                jnp.concatenate((joint_out["memory_delta"][None, ...], nonjoint_out["memory_delta"]), axis=0)
            ),
            "cross_stream_attention": cross_attention,
            "cross_stream_gates": cross_gates,
        }

        for name in (
            "motion_gate", "memory_gate", "controller_gate", "fusion_gate",
            "spatial_gate", "sms_gate", "sms_memory_delta", "l3_state_gate",
            "part_contrast_energy_ratio",
        ):
            if name in joint_out and name in nonjoint_out:
                result[name] = average_metric(name)
        return result

class DMGDL2DiagState(NamedTuple):
    momentum: Any
    projection: Any
    count: jax.Array

def dmgd_l2_diag():
    def init_fn(p):
        zeros = jax.tree_util.tree_map(jnp.zeros_like, p)
        return DMGDL2DiagState(
            momentum=zeros,
            projection=zeros,
            count=jnp.asarray(0, dtype=jnp.int32),
        )

    def update_fn(updates, state, params=None):
        del params
        new_momentum = jax.tree_util.tree_map(
            lambda m, g: DMGD_MOMENTUM * m + (1.0 - DMGD_MOMENTUM) * g,
            state.momentum,
            updates,
        )
        prediction = jax.tree_util.tree_map(
            lambda p, g: jnp.tanh(p) * g,
            state.projection,
            updates,
        )
        new_projection = jax.tree_util.tree_map(
            lambda p, g, pred, target: jnp.clip(
                p
                - DMGD_MEMORY_LR
                * (pred - jax.lax.stop_gradient(target))
                * g
                / (jnp.square(g) + DMGD_EPS),
                -DMGD_PROJECTION_CAP,
                DMGD_PROJECTION_CAP,
            ),
            state.projection,
            updates,
            prediction,
            new_momentum,
        )
        processed = jax.tree_util.tree_map(
            lambda g, pred: g + DMGD_MIX * pred,
            updates,
            prediction,
        )
        return processed, DMGDL2DiagState(
            momentum=new_momentum,
            projection=new_projection,
            count=state.count + jnp.asarray(1, dtype=jnp.int32),
        )

    return optax.GradientTransformation(init_fn, update_fn)

def scaled_schedule(total_steps, period):
    base = ns.make_schedule(total_steps)
    max_index = max(0, int(total_steps) - 1)
    def schedule(local_count):
        global_like = jnp.minimum(local_count * period, max_index)
        return base(global_like)
    return schedule

def merge_fragments(fragments):
    merged = {}
    for frag in fragments:
        flat = traverse_util.flatten_dict(frag)
        for path, value in flat.items():
            if path in merged:
                raise RuntimeError(f"Duplicate merged FAST path: {path}")
            merged[path] = value
    if set(merged.keys()) != set(flat_fast_params.keys()):
        raise RuntimeError("Merged FAST fragments do not reconstruct the original FAST tree.")

def evaluate_ema(
    ema,
    epoch,
):

    total_loss = 0.0

    total_correct = 0
    total_count = 0

    stream_correct = np.zeros(
        (
            4,
        ),

        dtype=np.int64,
    )


    last_fusion = None


    iterator = ns.batch_iterator(

        val_dataset,

        batch_size=GLOBAL_BATCH,

        shuffle=False,

        seed=SEED,

        drop_last=False,
    )


    pbar = tqdm(

        iterator,

        total=math.ceil(
            VAL_EXPECTED
            /
            GLOBAL_BATCH
        ),

        desc=
            f"E{epoch:02d} VAL",
    )


    for x_np, y_np in pbar:

        (
            x_np,
            y_np,
            valid_np,
            real_n,

        ) = pad_batch32(
            x_np,
            y_np,
        )


        (
            x,
            y,
            valid,

        ) = device_batch(

            x_np,
            y_np,
            valid_np,
        )


        (
            logits,
            stream_logits,
            fusion_weights,

        ) = eval_forward(

            ema,

            x,
        )


        logits_h = np.asarray(
            jax.device_get(
                logits
            )
        )


        stream_h = np.asarray(
            jax.device_get(
                stream_logits
            )
        )


        fusion_h = np.asarray(
            jax.device_get(
                fusion_weights
            )
        )


        labels_h = y_np[
            :real_n
        ]


        logits_real = logits_h[
            :real_n
        ]


        stream_real = stream_h[
            :real_n
        ]


        # CE without using padded samples.
        one_hot = np.eye(
            ns.CFG.num_classes,
            dtype=np.float32,
        )[
            labels_h
        ]


        smooth = (

            one_hot
            *
            (
                1.0
                -
                ns.CFG.label_smoothing
            )

            +

            ns.CFG.label_smoothing
            /
            ns.CFG.num_classes
        )


        # numerically stable host CE
        z = logits_real.astype(
            np.float64
        )


        z = (
            z
            -
            np.max(
                z,
                axis=-1,
                keepdims=True,
            )
        )


        log_probs = (

            z

            -

            np.log(

                np.sum(

                    np.exp(
                        z
                    ),

                    axis=-1,
                    keepdims=True,
                )
            )
        )


        ce = -np.sum(

            smooth
            *
            log_probs,

            axis=-1,
        )


        total_loss += float(
            np.sum(
                ce
            )
        )


        pred = np.argmax(

            logits_real,

            axis=-1,
        )


        total_correct += int(

            np.sum(
                pred
                ==
                labels_h
            )
        )


        for s in range(
            4
        ):

            stream_pred = np.argmax(

                stream_real[
                    :,
                    s,
                    :
                ],

                axis=-1,
            )


            stream_correct[
                s
            ] += int(

                np.sum(

                    stream_pred
                    ==
                    labels_h
                )
            )


        total_count += real_n


        last_fusion = fusion_h


        pbar.set_postfix(

            acc=
                f"{100.0 * total_correct / max(1,total_count):.2f}%"
        )


    if total_count != VAL_EXPECTED:

        raise RuntimeError(

            f"Validation count mismatch: "
            f"{total_count} != {VAL_EXPECTED}"
        )


    return {

        "loss":
            total_loss
            /
            total_count,

        "accuracy":
            total_correct
            /
            total_count,

        "correct":
            total_correct,

        "count":
            total_count,

        "stream_accuracy":
            (
                stream_correct
                /
                total_count
            ).tolist(),

        "fusion_weights":
            last_fusion.tolist(),
    }

def tree_numel(
    tree,
):

    return int(

        sum(

            x.size

            for x
            in jax.tree_util.tree_leaves(
                tree
            )
        )
    )


def tree_leaf_count(
    tree,
):

    return len(

        jax.tree_util.tree_leaves(
            tree
        )
    )


def merge_disjoint_trees(
    trees,
):

    merged = {}


    for tree in trees:

        flat = traverse_util.flatten_dict(
            tree
        )


        for path, value in flat.items():

            if path in merged:

                raise RuntimeError(
                    f"Duplicate parameter path: {path}"
                )


            merged[
                path
            ] = value


    return traverse_util.unflatten_dict(
        merged
    )


def tree_to_replicated(
    tree,
):

    return jax.tree_util.tree_map(

        lambda x:
            jax.device_put(
                x,
                REPLICATED,
            ),

        tree,
    )


def full_param_tree(
    fast_fragments,
    medium_params,
    slow_params,
    consolidate_params,
):

    fast = merge_fragments(
        fast_fragments
    )


    return merge_disjoint_trees(
        (
            fast,
            medium_params,
            slow_params,
            consolidate_params,
        )
    )

# ------------------------------------------------------------------------------------------
# CD-Former-style temporal reframing: LINEAR INTERPOLATION to exactly T32.
# No learned temporal embedding and no added model parameters.
# ------------------------------------------------------------------------------------------
def preprocess_keypoints_t32_interp(keypoints: np.ndarray) -> np.ndarray:
    x = ns.to_tmvc(keypoints)

    if x.shape[-1] == 2:
        zeros = np.zeros((*x.shape[:-1], 1), dtype=np.float32)
        x = np.concatenate([x, zeros], axis=-1)

    if x.shape[2] < ns.CFG.joints:
        pad = np.zeros(
            (x.shape[0], x.shape[1], ns.CFG.joints - x.shape[2], ns.CFG.coords),
            dtype=np.float32,
        )
        x = np.concatenate([x, pad], axis=2)
    else:
        x = x[:, :, :ns.CFG.joints, :ns.CFG.coords]

    person_energy = np.sum(np.abs(x), axis=(0, 2, 3))
    order = np.argsort(-person_energy)
    x = x[:, order]

    if x.shape[1] < ns.CFG.persons:
        pad = np.zeros(
            (x.shape[0], ns.CFG.persons - x.shape[1], ns.CFG.joints, ns.CFG.coords),
            dtype=np.float32,
        )
        x = np.concatenate([x, pad], axis=1)
    else:
        x = x[:, :ns.CFG.persons]

    total = int(x.shape[0])
    target = int(ns.CFG.frames)
    if total <= 0:
        x = np.zeros((target, ns.CFG.persons, ns.CFG.joints, ns.CFG.coords), dtype=np.float32)
    elif total == 1:
        x = np.repeat(x, target, axis=0)
    else:
        pos = np.linspace(0.0, float(total - 1), num=target, dtype=np.float32)
        lo = np.floor(pos).astype(np.int64)
        hi = np.minimum(lo + 1, total - 1)
        w = (pos - lo.astype(np.float32)).reshape(target, 1, 1, 1)
        x = (1.0 - w) * x[lo] + w * x[hi]

    center = x[:, 0:1, 0:1, :]
    valid_center = np.any(np.abs(center) > 1e-8, axis=-1, keepdims=True)
    x = np.where(valid_center, x - center, x)

    valid_joint = np.any(np.abs(x) > 1e-8, axis=-1, keepdims=True)
    x = np.where(valid_joint, x, 0.0)

    nonzero = np.abs(x) > 1e-8
    if np.any(nonzero):
        rms = float(np.sqrt(np.mean(np.square(x[nonzero]))) + 1e-6)
        x = x / rms

    x = np.nan_to_num(x, nan=0.0, posinf=0.0, neginf=0.0)
    return x.reshape(target, -1).astype(np.float32)

# 9. PAD FINAL PARTIAL BATCH TO GLOBAL B32
#
# Validity masking guarantees padded rows contribute:
#
#   0 loss
#   0 gradient
#   0 metric count
# ==========================================================================================

def pad_batch32(
    x,
    y,
):

    n = int(
        x.shape[
            0
        ]
    )


    if n > GLOBAL_BATCH:

        raise ValueError(
            f"Unexpected batch size: {n}"
        )


    valid = np.zeros(
        (
            GLOBAL_BATCH,
        ),

        dtype=np.float32,
    )


    valid[
        :n
    ] = 1.0


    if n < GLOBAL_BATCH:

        x_pad = np.zeros(

            (
                GLOBAL_BATCH
                -
                n,

                32,

                150,
            ),

            dtype=np.float32,
        )


        y_pad = np.zeros(

            (
                GLOBAL_BATCH
                -
                n,
            ),

            dtype=np.int32,
        )


        x = np.concatenate(
            (
                x,
                x_pad,
            ),
            axis=0,
        )


        y = np.concatenate(
            (
                y,
                y_pad,
            ),
            axis=0,
        )


    return (

        x.astype(
            np.float32
        ),

        y.astype(
            np.int32
        ),

        valid,

        n,
    )

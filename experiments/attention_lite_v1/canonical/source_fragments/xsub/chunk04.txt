def tree_numel(
    tree,
):

    return int(

        sum(

            x.size

            for x
            in jax.tree_util.tree_leaves(
                tree
            )
        )
    )


def tree_leaf_count(
    tree,
):

    return len(

        jax.tree_util.tree_leaves(
            tree
        )
    )


def merge_disjoint_trees(
    trees,
):

    merged = {}


    for tree in trees:

        flat = traverse_util.flatten_dict(
            tree
        )


        for path, value in flat.items():

            if path in merged:

                raise RuntimeError(
                    f"Duplicate parameter path: {path}"
                )


            merged[
                path
            ] = value


    return traverse_util.unflatten_dict(
        merged
    )


def tree_to_replicated(
    tree,
):

    return jax.tree_util.tree_map(

        lambda x:
            jax.device_put(
                x,
                REPLICATED,
            ),

        tree,
    )


def full_param_tree(
    fast_fragments,
    medium_params,
    slow_params,
    consolidate_params,
):

    fast = merge_fragments(
        fast_fragments
    )


    return merge_disjoint_trees(
        (
            fast,
            medium_params,
            slow_params,
            consolidate_params,
        )
    )

# 9. PAD FINAL PARTIAL BATCH TO GLOBAL B32
#
# Validity masking guarantees padded rows contribute:
#
#   0 loss
#   0 gradient
#   0 metric count
# ==========================================================================================

def pad_batch32(
    x,
    y,
):

    n = int(
        x.shape[
            0
        ]
    )


    if n > GLOBAL_BATCH:

        raise ValueError(
            f"Unexpected batch size: {n}"
        )


    valid = np.zeros(
        (
            GLOBAL_BATCH,
        ),

        dtype=np.float32,
    )


    valid[
        :n
    ] = 1.0


    if n < GLOBAL_BATCH:

        x_pad = np.zeros(

            (
                GLOBAL_BATCH
                -
                n,

                16,

                150,
            ),

            dtype=np.float32,
        )


        y_pad = np.zeros(

            (
                GLOBAL_BATCH
                -
                n,
            ),

            dtype=np.int32,
        )


        x = np.concatenate(
            (
                x,
                x_pad,
            ),
            axis=0,
        )


        y = np.concatenate(
            (
                y,
                y_pad,
            ),
            axis=0,
        )


    return (

        x.astype(
            np.float32
        ),

        y.astype(
            np.int32
        ),

        valid,

        n,
    )


def device_batch(

def head_outputs(
    fusion_logits,
    joint_active,
    nonjoint_active,
    joint_target,
    nonjoint_target,
):

    (
        joint_logits,
        joint_prediction,

    ) = joint_active


    (
        nonjoint_logits,
        nonjoint_prediction,

    ) = nonjoint_active


    stream_logits = jnp.swapaxes(

        jnp.concatenate(

            (

                joint_logits[
                    None,
                    ...
                ],

                nonjoint_logits,
            ),

            axis=0,
        ),

        0,
        1,
    )


    fusion_weights = jax.nn.softmax(
        fusion_logits
    )


    logits = jnp.einsum(

        "s,bsc->bc",

        fusion_weights,

        stream_logits,
    )


    stream_prediction = jnp.swapaxes(

        jnp.concatenate(

            (

                joint_prediction[
                    None,
                    ...
                ],

                nonjoint_prediction,
            ),

            axis=0,
        ),

        0,
        1,
    )


    stream_target = jax.lax.stop_gradient(

        jnp.swapaxes(

            jnp.concatenate(

                (

                    joint_target[
                        None,
                        ...
                    ],

                    nonjoint_target,
                ),

                axis=0,
            ),

            0,
            1,
        )
    )


    return (

        logits,

        stream_logits,

        fusion_weights,

        stream_prediction,

        stream_target,
    )


def masked_head_loss(
    fusion_logits,
    joint_active,
    nonjoint_active,
    joint_target,
    nonjoint_target,
    labels,
    valid,
):

    (
        logits,
        stream_logits,
        fusion_weights,
        stream_prediction,
        stream_target,

    ) = head_outputs(

        fusion_logits,

        joint_active,
        nonjoint_active,

        joint_target,
        nonjoint_target,
    )


    valid = valid.astype(
        jnp.float32
    )


    denom = jnp.maximum(

        jnp.sum(
            valid
        ),

        1.0,
    )


    # --------------------------------------------------------------------------------------
    # FUSED CE
    # --------------------------------------------------------------------------------------

    one_hot = jax.nn.one_hot(

        labels,

        ns.CFG.num_classes,
    )


    smooth = (

        one_hot
        *
        (
            1.0
            -
            ns.CFG.label_smoothing
        )

        +

        ns.CFG.label_smoothing
        /
        ns.CFG.num_classes
    )


    fused_ce_each = optax.softmax_cross_entropy(

        logits,

        smooth,
    )


    fused_ce = (

        jnp.sum(

            fused_ce_each
            *
            valid
        )

        /

        denom
    )


    # --------------------------------------------------------------------------------------
    # AUX STREAM CE
    # [B,4,C]
    # --------------------------------------------------------------------------------------

    smooth_stream = smooth[
        :,
        None,
        :
    ]


    stream_ce_each = optax.softmax_cross_entropy(

        stream_logits,

        smooth_stream,
    )


    stream_ce_each = jnp.mean(

        stream_ce_each,

        axis=1,
    )


    stream_ce = (

        jnp.sum(

            stream_ce_each
            *
            valid
        )

        /

        denom
    )


    # --------------------------------------------------------------------------------------
    # PREDICTIVE
    #
    # prediction t predicts target t+1
    # --------------------------------------------------------------------------------------

    pred_now = stream_prediction[
        :,
        :,
        :-1,
        :
    ]


    target_next = jax.lax.stop_gradient(

        stream_target[
            :,
            :,
            1:,
            :
        ]
    )


    pred_each = jnp.mean(

        jnp.square(

            pred_now
            -
            target_next
        ),

        axis=(
            1,
            2,
            3,
        ),
    )


    predictive = (

        jnp.sum(

            pred_each
            *
            valid
        )

        /

        denom
    )


    total = (

        fused_ce

        +

        m4.AUX_STREAM_CE_WEIGHT
        *
        stream_ce

        +

        ns.CFG.predictive_loss_weight
        *
        predictive
    )


    predicted_class = jnp.argmax(

        logits,

        axis=-1,
    )


    correct = jnp.sum(

        (
            predicted_class
            ==
            labels
        ).astype(
            jnp.float32
        )

        *
        valid
    )


    # Stream correct counts.
    stream_pred_class = jnp.argmax(

        stream_logits,

        axis=-1,
    )


    stream_correct = jnp.sum(

        (
            stream_pred_class
            ==
            labels[
                :,
                None
            ]
        ).astype(
            jnp.float32
        )

        *
        valid[
            :,
            None
        ],

        axis=0,
    )


    aux = (

        fused_ce,
        stream_ce,
        predictive,

        correct,
        denom,

        fusion_weights,

        stream_correct,
    )


    return (
        total,
        aux,
    )

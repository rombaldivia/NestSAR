class DMGDL2DiagState(NamedTuple):
    momentum: Any
    projection: Any
    count: jax.Array

def dmgd_l2_diag():
    def init_fn(p):
        zeros = jax.tree_util.tree_map(jnp.zeros_like, p)
        return DMGDL2DiagState(
            momentum=zeros,
            projection=zeros,
            count=jnp.asarray(0, dtype=jnp.int32),
        )

    def update_fn(updates, state, params=None):
        del params
        new_momentum = jax.tree_util.tree_map(
            lambda m, g: DMGD_MOMENTUM * m + (1.0 - DMGD_MOMENTUM) * g,
            state.momentum,
            updates,
        )
        prediction = jax.tree_util.tree_map(
            lambda p, g: jnp.tanh(p) * g,
            state.projection,
            updates,
        )
        new_projection = jax.tree_util.tree_map(
            lambda p, g, pred, target: jnp.clip(
                p
                - DMGD_MEMORY_LR
                * (pred - jax.lax.stop_gradient(target))
                * g
                / (jnp.square(g) + DMGD_EPS),
                -DMGD_PROJECTION_CAP,
                DMGD_PROJECTION_CAP,
            ),
            state.projection,
            updates,
            prediction,
            new_momentum,
        )
        processed = jax.tree_util.tree_map(
            lambda g, pred: g + DMGD_MIX * pred,
            updates,
            prediction,
        )
        return processed, DMGDL2DiagState(
            momentum=new_momentum,
            projection=new_projection,
            count=state.count + jnp.asarray(1, dtype=jnp.int32),
        )

    return optax.GradientTransformation(init_fn, update_fn)

def scaled_schedule(total_steps, period):
    base = ns.make_schedule(total_steps)
    max_index = max(0, int(total_steps) - 1)
    def schedule(local_count):
        global_like = jnp.minimum(local_count * period, max_index)
        return base(global_like)
    return schedule

def merge_fragments(fragments):
    merged = {}
    for frag in fragments:
        flat = traverse_util.flatten_dict(frag)
        for path, value in flat.items():
            if path in merged:
                raise RuntimeError(f"Duplicate merged FAST path: {path}")
            merged[path] = value
    if set(merged.keys()) != set(flat_fast_params.keys()):
        raise RuntimeError("Merged FAST fragments do not reconstruct the original FAST tree.")
    return traverse_util.unflatten_dict(merged)

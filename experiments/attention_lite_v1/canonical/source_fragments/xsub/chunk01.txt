    value = float(np.clip(value, 1e-6, 1.0 - 1e-6))
    return float(np.log(value / (1.0 - value)))

# 3. HOPE-ATTENTION LITE TEMPORAL BLOCK
# ==========================================================================================

class HopeAttentionLiteBlock(nn.Module):

    """
    Drop-in replacement for M4 MultiHeadAdaptiveMemoryBlock.

    Sequence mechanism:

        x
        ↓
    causal local depthwise Conv1D(k=4)
        ↓
    global CAUSAL softmax self-attention
        ↓
    residual projection
        ↓
    sequential CMS f1→f2→f4→f8

    Attention width is intentionally 64 while the outer model remains D128.
    """

    model_dim: int
    memory_dim: int

    dropout: float
    residual_scale: float

    initial_eta: float
    initial_alpha: float

    heads: int = ATTENTION_HEADS


    @nn.compact
    def __call__(
        self,
        x: jnp.ndarray,
        training: bool,
    ):

        if x.ndim != 3:

            raise ValueError(
                f"Expected [B,T,D], got {x.shape}"
            )


        if self.model_dim != 128:

            raise ValueError(
                "HOPE-Attention-Lite v1 is locked to D128."
            )


        if self.heads != ATTENTION_HEADS:

            raise ValueError(
                f"Expected {ATTENTION_HEADS} attention heads."
            )


        batch, time_steps, _ = x.shape


        # ==================================================================================
        # A. INPUT NORMALIZATION
        # ==================================================================================

        h = nn.LayerNorm(
            name="input_norm"
        )(x)


        # ==================================================================================
        # B. HOPE LOCAL CAUSAL CONVOLUTION — k=4
        #
        # left pad 3:
        #
        # output[t] only sees:
        #   t, t-1, t-2, t-3
        #
        # depthwise:
        #   groups = D128
        # ==================================================================================

        padded = jnp.pad(

            h,

            (
                (0, 0),

                (
                    LOCAL_KERNEL - 1,
                    0,
                ),

                (0, 0),
            ),
        )


        local = nn.Conv(

            features=self.model_dim,

            kernel_size=(
                LOCAL_KERNEL,
            ),

            strides=(
                1,
            ),

            padding="VALID",

            feature_group_count=self.model_dim,

            use_bias=True,

            kernel_init=nn.initializers.xavier_uniform(),

            name="hope_local_conv",

        )(
            padded
        )


        local = jax.nn.silu(
            local
        )


        local_gate_logit = self.param(

            "hope_local_conv_gate_logit",

            lambda key, shape:

                jnp.full(

                    shape,

                    fidelity_core._safe_logit(
                        LOCAL_CONV_GATE_INIT
                    ),

                    dtype=jnp.float32,
                ),

            (
                1,
            ),
        )


        local_gate = jax.nn.sigmoid(

            local_gate_logit

        )[0]


        x = nn.LayerNorm(

            name="hope_local_conv_output_norm"

        )(
            x
            +
            local_gate
            *
            local
        )


        # ==================================================================================
        # C. GLOBAL CAUSAL SOFTMAX ATTENTION
        #
        # Dmodel   = 128
        # Dattn    = 64
        # heads    = 4
        # head_dim = 16
        # ==================================================================================

        a = nn.LayerNorm(
            name="attention_norm"
        )(x)


        def project(
            name: str,
        ) -> jnp.ndarray:

            z = nn.Dense(

                features=ATTENTION_DIM,

                use_bias=False,

                kernel_init=
                    nn.initializers.xavier_uniform(),

                name=name,

            )(
                a
            )


            return z.reshape(

                batch,

                time_steps,

                ATTENTION_HEADS,

                HEAD_DIM,
            )


        q = project(
            "attention_query"
        )

        k = project(
            "attention_key"
        )

        v = project(
            "attention_value"
        )


        # [B,H,T,T]
        scores = jnp.einsum(

            "bthd,bshd->bhts",

            q,
            k,
        )


        scores = (

            scores

            *

            jax.lax.rsqrt(

                jnp.asarray(

                    float(
                        HEAD_DIM
                    ),

                    dtype=scores.dtype,
                )
            )
        )


        # ------------------------------------------------------------------
        # Causal global attention.
        #
        # Query position t may attend only source positions <= t.
        # ------------------------------------------------------------------

        positions = jnp.arange(
            time_steps
        )


        causal_mask = (

            positions[
                None,
                None,
                :,
                None
            ]

            >=

            positions[
                None,
                None,
                None,
                :
            ]
        )


        scores = jnp.where(

            causal_mask,

            scores,

            jnp.asarray(
                -1.0e30,
                dtype=scores.dtype,
            ),
        )


        attention_weights = jax.nn.softmax(

            scores,

            axis=-1,
        )


        # [B,T,H,Dh]
        attended = jnp.einsum(

            "bhts,bshd->bthd",

            attention_weights,

            v,
        )


        # ------------------------------------------------------------------
        # Retain the same kind of per-head learned residual gate used by the
        # previous memory interface.
        # ------------------------------------------------------------------

        attention_gate_logit = self.param(

            "memory_gate_logit",

            lambda key, shape:

                jnp.full(

                    shape,

                    fidelity_core._safe_logit(
                        ATTENTION_GATE_INIT
                    ),

                    dtype=jnp.float32,
                ),

            (
                ATTENTION_HEADS,
            ),
        )


        attention_gate = jax.nn.sigmoid(

            attention_gate_logit
        )


        attended = (

            attended

            *

            attention_gate[
                None,
                None,
                :,
                None,
            ]
        )


        attended = attended.reshape(

            batch,

            time_steps,

            ATTENTION_DIM,
        )


        # 64 -> 128
        context = nn.Dense(

            features=self.model_dim,

            use_bias=True,

            kernel_init=
                nn.initializers.xavier_uniform(),

            name="readout",

        )(
            attended
        )


        context = nn.Dropout(

            rate=self.dropout,

            name="context_dropout",

        )(
            context,

            deterministic=not training,
        )


        scaled_context = (

            self.residual_scale

            *

            context
        )


        x = nn.LayerNorm(

            name="memory_residual_norm"

        )(
            x
            +
            scaled_context
        )


        # ==================================================================================
        # D. HOPE SEQUENTIAL CONTINUUM MEMORY SYSTEM
        #
        # EXACT SAME NAMES AS FIDELITY:
        #
        #     cms_f1
        #        ↓
        #     cms_f2
        #        ↓
        #     cms_f4
        #        ↓
        #     cms_f8
        #
        # therefore Fidelity's _path_to_tier() still maps:
        #
        #     f1 -> FAST
        #     f2 -> MEDIUM
        #     f4 -> SLOW
        #     f8 -> CONSOLIDATE
        # ==================================================================================

        output = x


        for cms_frequency in (
            1,
            2,
            4,
            8,
        ):

            cms_h = nn.LayerNorm(

                name=
                    f"cms_f{cms_frequency}_norm"

            )(
                output
            )


            cms_h = nn.Dense(

                features=CMS_BOTTLENECK,

                use_bias=True,

                kernel_init=
                    nn.initializers.xavier_uniform(),

                name=
                    f"cms_f{cms_frequency}_in",

            )(
                cms_h
            )


            cms_h = jax.nn.silu(
                cms_h
            )


            cms_h = nn.Dropout(

                rate=self.dropout,

                name=
                    f"cms_f{cms_frequency}_dropout",

            )(
                cms_h,

                deterministic=not training,
            )


            cms_delta = nn.Dense(

                features=self.model_dim,

                use_bias=True,

                kernel_init=
                    nn.initializers.xavier_uniform(),

                name=
                    f"cms_f{cms_frequency}_out",

            )(
                cms_h
            )


            cms_gate_logit = self.param(

                f"cms_f{cms_frequency}_gate_logit",

                lambda key, shape:

                    jnp.full(

                        shape,

                        fidelity_core._safe_logit(
                            0.05
                        ),

                        dtype=jnp.float32,
                    ),

                (
                    1,
                ),
            )


            cms_gate = jax.nn.sigmoid(

                cms_gate_logit

            )[0]


            output = (

                output

                +

                cms_gate
                *
                cms_delta
            )


        output = nn.LayerNorm(

            name="output_norm"

        )(
            output
        )


        # ==================================================================================
        # E. COMPATI

BILITY DIAGNOSTIC
        #
        # This is no longer a recurrent-memory state delta.
        #
        # For compatibility with the M4 trainer we return the RMS magnitude
        # of the attention residual per token:
        #
        #     [B,T]
        #
        # It is diagnostic only; it does NOT enter the classification loss.
        # ==================================================================================

        attention_residual_delta = jnp.sqrt(

            jnp.mean(

                jnp.square(
                    scaled_context
                ),

                axis=-1,
            )

            +

            1.0e-12
        )


        return (

            output,

            scaled_context,

            attention_residual_delta,

            jnp.mean(
                attention_gate
            ),
        )

# ==========================================================================================
# 6. TOP-LEVEL MODEL
# ==========================================================================================

class NestSARHOPEAttentionLiteVec31(
    nn.Module
):

    num_classes: int

    model_dim: int
    memory_dim: int

    dropout: float

    memory_residual_scale: float

    initial_eta: float
    initial_alpha: float

    frame_blocks: int = 2
    chunk_blocks: int = 2
    clip_blocks: int = 2

    controller_blocks: int = 2

    chunk_size: int = 4
    clip_size: int = 8

    controller_rank: int = 32


    @nn.compact
    def __call__(
        self,
        x: jnp.ndarray,
        training: bool,
    ):

        if x.shape[1] % self.chunk_size:

            raise ValueError(
                "frames must be divisible by chunk_size"
            )


        if x.shape[1] % self.clip_size:

            raise ValueError(
                "frames must be divisible by clip_size"
            )


        streams = m4.build_four_streams(
            x
        )


        joint_xyz = streams[
            "joint"
        ]


        geometry = m4.geometric_features(
            joint_xyz
        )


        # ==================================================================================
        # JOINT
        # ==================================================================================

        joint_out = StreamBackbone(

            stream_name="joint",

            num_classes=self.num_classes,

            model_dim=self.model_dim,

            memory_dim=self.memory_dim,

            dropout=self.dropout,

            memory_residual_scale=
                self.memory_residual_scale,

            initial_eta=self.initial_eta,

            initial_alpha=self.initial_alpha,

            frame_blocks=self.frame_blocks,

            chunk_blocks=self.chunk_blocks,

            clip_blocks=self.clip_blocks,

            controller_blocks=
                self.controller_blocks,

            chunk_size=self.chunk_size,

            clip_size=self.clip_size,

            controller_rank=
                self.controller_rank,

            use_part_attention=False,

            name="stream_joint",

        )(
            joint_xyz,

            joint_xyz,

            geometry,

            training,
        )


        # ==================================================================================
        # NONJOINT ×3
        # ==================================================================================

        nonjoint_xyz = jnp.stack(

            (
                streams[
                    "bone"
                ],

                streams[
                    "joint_motion"
                ],

                streams[
                    "bone_motion"
                ],
            ),

            axis=0,
        )


        # Spatial behavior only distinguishes joint vs non-joint.
        # "bone" is therefore the canonical static non-joint name.
        nonjoint_out = VectorizedNonJointBackbone(

            stream_name="bone",

            num_classes=self.num_classes,

            model_dim=self.model_dim,

            memory_dim=self.memory_dim,

            dropout=self.dropout,

            memory_residual_scale=
                self.memory_residual_scale,

            initial_eta=self.initial_eta,

            initial_alpha=self.initial_alpha,

            frame_blocks=self.frame_blocks,

            chunk_blocks=self.chunk_blocks,

            clip_blocks=self.clip_blocks,

            controller_blocks=
                self.controller_blocks,

            chunk_size=self.chunk_size,

            clip_size=self.clip_size,

            controller_rank=
                self.controller_rank,

            use_part_attention=False,

            name="stream_nonjoint_bank",

        )(
            nonjoint_xyz,

            joint_xyz,

            geometry,

            training,
        )


        # ==================================================================================
        # EXACT FOUR-STREAM FUSION
        # ==================================================================================

        stream_logits = jnp.swapaxes(

            jnp.concatenate(

                (
                    joint_out[
                        "logits"
                    ][
                        None,
                        ...
                    ],

                    nonjoint_out[
                        "logits"
                    ],
                ),

                axis=0,
            ),

            0,
            1,
        )


        fusion_logits = self.param(

            "stream_fusion_logits",

            nn.initializers.zeros,

            (
                NUM_STREAMS,
            ),
        )


        fusion_weights = jax.nn.softmax(

            fusion_logits
        )


        logits = jnp.einsum(

            "s,bsc->bc",

            fusion_weights,

            stream_logits,
        )


        # ==================================================================================
        # PREDICTIVE STREAMS
        # ==================================================================================

        stream_prediction = jnp.swapaxes(

            jnp.concatenate(

                (
                    joint_out[
                        "prediction"
                    ][
                        None,
                        ...
                    ],

                    nonjoint_out[
                        "prediction"
                    ],
                ),

                axis=0,
            ),

            0,
            1,
        )


        stream_motion_target = jnp.swapaxes(

            jnp.concatenate(

                (
                    joint_out[
                        "motion_target"
                    ][
                        None,
                        ...
                    ],

                    nonjoint_out[
                        "motion_target"
                    ],
                ),

                axis=0,
            ),

            0,
            1,
        )


        prediction = jnp.einsum(

            "s,bstd->btd",

            fusion_weights,

            stream_prediction,
        )


        motion_target = jnp.einsum(

            "s,bstd->btd",

            fusion_weights,

            stream_motion_target,
        )


        # ==================================================================================
        # METRIC AGGREGATION
        # ==================================================================================

        def average_metric(
            name: str,
        ):

            joint_mean = jnp.mean(

                joint_out[
                    name
                ]
            )


            value = nonjoint_out[
                name
            ]


            if value.ndim == 1:

                nonjoint_means = value

            else:

                nonjoint_means = jnp.mean(

                    value,

                    axis=tuple(

                        range(
                            1,
                            value.ndim,
                        )
                    ),
                )


            values = jnp.concatenate(

                (
                    joint_mean[
                        None
                    ],

                    nonjoint_means.reshape(
                        (
                            NUM_NONJOINT,
                        )
                    ),
                ),

                axis=0,
            )


            return jnp.mean(
                values
            )


        result = {

            "logits":
                logits,

            "stream_logits":
                stream_logits,

            "fusion_weights":
                fusion_weights,

            "prediction":
                prediction,

            "motion_target":
                motion_target,

            "stream_prediction":
                stream_prediction,

            "stream_motion_target":
                stream_motion_target,

            "memory_delta":
                jnp.mean(

                    jnp.concatenate(

                        (
                            joint_out[
                                "memory_delta"
                            ][
                                None,
                                ...
                            ],

                            nonjoint_out[
                                "memory_delta"
                            ],
                        ),

                        axis=0,
                    )
                ),
        }


        candidate_metrics = (

            "motion_gate",

            "memory_gate",

            "controller_gate",

            "fusion_gate",

            "spatial_gate",

            "sms_gate",

            "sms_memory_delta",

            "l3_state_gate",

            "part_contrast_energy_ratio",
        )


        for name in candidate_metrics:

            if (
                name in joint_out
                and
                name in nonjoint_out
            ):

                result[
                    name
                ] = average_metric(
                    name
                )


        return result

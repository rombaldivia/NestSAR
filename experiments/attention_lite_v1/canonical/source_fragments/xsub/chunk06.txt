        raise RuntimeError(

            f"Validation count mismatch: "
            f"{total_count} != {VAL_EXPECTED}"
        )


    return {

        "loss":
            total_loss
            /
            total_count,

        "accuracy":
            total_correct
            /
            total_count,

        "correct":
            total_correct,

        "count":
            total_count,

        "stream_accuracy":
            (
                stream_correct
                /
                total_count
            ).tolist(),

        "fusion_weights":
            last_fusion.tolist(),
    }
